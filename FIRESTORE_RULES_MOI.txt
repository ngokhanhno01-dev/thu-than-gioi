rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /saves/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

    match /nicks/{key} {
      allow get: if true;
      allow list: if false;
      allow create: if request.auth != null
        && key.matches('^[a-z0-9]{3,16}$')
        && request.resource.data.keys().hasOnly(['uid', 'nick'])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.nick is string
        && request.resource.data.nick.size() >= 3 && request.resource.data.nick.size() <= 16;
      allow update: if false;
      // MỚI: chủ sở hữu được xóa tên của mình (nhả tên cũ khi đổi biệt danh)
      allow delete: if request.auth != null && resource.data.uid == request.auth.uid;
    }

    match /profiles/{uid} {
      allow read: if true;
      // MỚI: thêm "update" để đổi biệt danh được (điều kiện giữ nguyên như create)
      allow create, update: if request.auth != null && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['nick', 'nickKey', 't'])
        && request.resource.data.nick is string
        && request.resource.data.nick.size() >= 3 && request.resource.data.nick.size() <= 16
        && request.resource.data.nickKey.matches('^[a-z0-9]{3,16}$')
        && getAfter(/databases/$(database)/documents/nicks/$(request.resource.data.nickKey)).data.uid == request.auth.uid
        && request.resource.data.t == request.time;
      allow delete: if false;
    }

    // BẠN BÈ - BƯỚC 1: thẻ thú công khai. Ai cũng đọc được, mỗi người chỉ ghi được thẻ của chính mình.
    match /cards/{uid} {
      allow read: if true;
      allow create, update: if request.auth != null && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['nick','lv','st','sp','bl','pi','up','q','stats','eq','t'])
        && request.resource.data.nick is string && request.resource.data.nick.size() <= 20
        && request.resource.data.lv is int && request.resource.data.lv >= 1 && request.resource.data.lv <= 100
        && request.resource.data.st is int && request.resource.data.st >= 0 && request.resource.data.st <= 9
        && request.resource.data.sp is int && request.resource.data.sp >= 0 && request.resource.data.sp <= 16
        && request.resource.data.bl in ['ho','vu','long','tuoc','lan','hu']
        && request.resource.data.pi is number && request.resource.data.pi >= 0 && request.resource.data.pi <= 4
        && request.resource.data.up is int && request.resource.data.up >= 0 && request.resource.data.up <= 10
        && request.resource.data.q is int && request.resource.data.q >= 0 && request.resource.data.q <= 4
        && request.resource.data.stats is map
        && request.resource.data.eq is list && request.resource.data.eq.size() == 4
        && request.resource.data.t == request.time;
      allow delete: if request.auth != null && request.auth.uid == uid;
    }

    // ---- Hàm phụ: ghép 2 uid theo thứ tự chữ cái (dùng làm id tình bạn) ----
    function pairId(x, y) {
      return x < y ? x + '_' + y : y + '_' + x;
    }

    // BẠN BÈ - BƯỚC 2a: lời mời kết bạn. Id = "uidNgườiMời_uidNgườiNhận".
    // Chỉ 2 người liên quan đọc được. Chỉ tạo được lời mời mang tên chính mình, và chưa là bạn.
    match /friendreqs/{id} {
      allow read: if request.auth != null
        && (resource.data.from == request.auth.uid || resource.data.to == request.auth.uid);
      allow create: if request.auth != null
        && request.resource.data.keys().hasOnly(['from', 'to', 't'])
        && request.resource.data.from == request.auth.uid
        && request.resource.data.to is string
        && request.resource.data.to.size() >= 1 && request.resource.data.to.size() <= 128
        && request.resource.data.to != request.auth.uid
        && id == request.auth.uid + '_' + request.resource.data.to
        && request.resource.data.t == request.time
        && !exists(/databases/$(database)/documents/friendships/$(pairId(request.auth.uid, request.resource.data.to)));
      allow update: if false;
      // người mời hủy, hoặc người nhận từ chối / đã chấp nhận
      allow delete: if request.auth != null
        && (resource.data.from == request.auth.uid || resource.data.to == request.auth.uid);
    }

    // BẠN BÈ - BƯỚC 2b: tình bạn. Id = 2 uid xếp theo chữ cái, nối bằng "_".
    // Chỉ tạo được khi NGƯỜI KIA đã gửi lời mời cho mình (friendreqs/{họ}_{mình} đang tồn tại).
    match /friendships/{id} {
      allow read: if request.auth != null && request.auth.uid in resource.data.users;
      allow create: if request.auth != null
        && request.resource.data.keys().hasOnly(['users', 't'])
        && request.resource.data.users is list
        && request.resource.data.users.size() == 2
        && request.auth.uid in request.resource.data.users
        && request.resource.data.users[0] < request.resource.data.users[1]
        && id == request.resource.data.users[0] + '_' + request.resource.data.users[1]
        && request.resource.data.t == request.time
        && exists(/databases/$(database)/documents/friendreqs/$(
             (request.resource.data.users[0] == request.auth.uid
               ? request.resource.data.users[1]
               : request.resource.data.users[0]) + '_' + request.auth.uid));
      allow update: if false;
      // một trong hai người hủy kết bạn
      allow delete: if request.auth != null && request.auth.uid in resource.data.users;
    }
  }
}
